//.bmxon
// JSON互換のファイル形式

// 型の定義
// bool		->	true または false
// number	-> 倍精度浮動小数点数値(JSONの規格により、infinity及びNaNは禁止)
// string	-> 文字列(""で囲む)。stringが指定されるべきプロパティにnumberが記述されている場合、それを単純に文字列に変換する(例: 1 -> "1"、エラーにはしない)
// Enum		-> number または その値と1:1対応するstring。文字列としては半角英数及び_のみ使用可能で、大文字と小文字を区別しない
// Rational -> number または "n/d"形式の文字列。解釈器はラウンドトリップを保証しなければならない
// Position	-> 時間位置を表す文字列。詳細は後述
// その他(大文字始まりの型名)	-> 独自型(Objectの派生)

// 特殊な型記法
// T[]		-> Tの配列
// T{}		-> Tを値とする連想配列
// ref T	-> T または Tを格納した個別のファイルへの相対パスを示す文字列
// T1 < T2	-> T2を継承した型T1(T1はT2が持つ全てのプロパティを持つ)

// 値が解釈不能な場合(例: Rationalとして解釈できない文字列)、またはそのプロパティが存在しない場合、原則としてデフォルト値(null, false, 0, "")を採用する。

// refにファイル名を指定する場合、拡張子を含める。そのファイル存在しない場合はそのプロパティが省略されていると見なす。
// 解釈器は、この拡張子を独自に変換してもよい。例えばbm3on上で"hoge.wav"と指定されているファイルを、代わりに"hoge.ogg"を指定していると見なしてもよい。
// 解釈器は、拡張子を除いたファイル名を独自に変換してはならない。例えばbm3on上で"hoge.wav"と指定されているファイルを、代わりに"fuga.wav"を指定していると見なしてはならない。

// Position の記法
// "n/d"	-> 楽曲先頭を 0 とした場合の絶対位置(time_base依存)
// "#m:n/d"	-> 小節番号m内の、n/dに相当する位置(time_base依存)。mは0以上の整数。":"が区切り文字として要求され、"#"は省略してもよい
//
// 時刻表現(Conductor.stopを含む)は全音符(四分音符×4)の長さを単位とし、全音符の長さは header.time_base によって規定される(省略時は 1)。
// time_base = 1 の場合、四分音符1個分の長さ/オフセットは "1/4" となる。
// time_base = 240 の場合、四分音符1個分の長さ/オフセットは "60/1" となり、これは単に数値の 60 と書くこともできる。

// 全てのオブジェクトは、"note"という名前の文字列型プロパティを持つことができる。これはそのオブジェクトについてのメモ書きで、ゲームプレイには影響しない

// ルートデータ
object BmxonData
{
	"header"	: HeaderInfo,		// ヘッダ情報。存在しない場合は不正なファイルと見なす
	
	"bars"		: string{},			// 小節番号(0以上の整数)をキー、"n/d"形式で拍子を表す文字列を値とする連想配列。初期値は4/4。拍子が未指定の小節は、その直前の小節の拍子を継承する
	"sounds"	: (ref Audio){},	// ID(任意の整数)をキーとする Audio の連想配列
	"images"	: (ref Image){},	// ID(任意の整数)をキーとする Image の連想配列。IDは sounds と重複してもよい

	"conductor"		: Conductor[],	// 指揮者オブジェクトのリスト
	"play_notes"	: Note[],		// 演奏オブジェクトのリスト
	"meta_notes"	: MetaNote[],	// メタオブジェクトのリスト

	"flow_data"	: (ref BmxonData){}	// ID(任意の整数)をキーとするランダム分岐の内容の連想配列。その内容はこのファイル内に記述してもいいし、別ファイルへの参照にしてもいい
	"flows"		: FlowInfo[],		// ランダム分岐情報のリスト
}

object HeaderInfo
{
	"genre"		: string,	// 楽曲のジャンル
	"title"		: string,	// 題名。省略不可能。不正値(省略/解釈不能/空文字列)の場合は不正なファイルと見なす
	"subtitle"	: string,	// 副題。差分名のために使ってもよい。
	"artist"	: string,	// 作成者。省略不可能。不正値(省略/解釈不能/空文字列)の場合は不正なファイルと見なす
	"subartist"	: string,	// 補助作成者。複数いる場合は"/"で区切るなどする
	"comment"	: string,	// 楽曲コメント
	
	"time_base"		: Rational,	// 時間単位。全音符1つの長さをこの数値とする。各ノートの"p"及び"l"に影響する。省略時は 1
	"display_bpm"	: string,	// 選曲画面などのためのテンポ表記。省略時はconductorから推測する。p=0のテンポ指定が無い場合、楽曲開始時のテンポは120bpmと見なす

	"chart_type"	: string,	// 譜面の形式。bmsonのmode_hintと同様の内容を想定する。推奨値は後述するが、省略時の値も含め、解釈器はこれに従わなくてもよい
	"ln_type"		: Enum,		// ロングノート形式。推奨値は後述するが、省略時の値も含め、解釈器はこれに従わなくてもよい
	"difficulty"	: Enum,		// 選曲画面などのための難度表記。推奨値は後述するが、省略時の値も含め、解釈器はこれに従わなくてもよい
	"level"			: number,	// 選曲画面などのためのプレイレベル。推奨値は後述するが、省略時の値も含め、解釈器はこれに従わなくてもよい
	"judge"			: number,	// 判定幅。推奨値は後述するが、省略時の値も含め、解釈器はこれに従わなくてもよい
	"total"			: number,	// 全てのノーツを最高判定で処理した場合に回復するゲージの量。推奨値は後述するが、省略時の値も含め、解釈器はこれに従わなくてもよい

	"preview"	: ref Audio,	// プレビュー音声
	"jacket"	: ref Image,	// ジャケット画像
	"banner"	: ref Image,	// バナー画像(横長の画像)
	"splash"	: ref Image,	// ロード中等に表示する画像(特殊フォントによるタイトル画像など)
	"background": ref Image,	//  プレイ画面等で表示する背景画像(大きな画像)
}

// HeaderInfo.chart_type の推奨値
enum ChartType
{
	"beat-5k", "beat-10k",	// いわゆる5鍵
	"beat-7k", "beat-14k",	// いわゆる7鍵
	"popn-3k",	// Pop系バトル
	"popn-5k", "popn-9k"	// Pop系
	"generic-24k", "generic-48k"	// キーボード系
}

// HeaderInfo.ln_type の推奨解釈
enum LnType
{
	0 = "Auto",	// 自動判定
	1 = "LN",	// 旧式ロングノート
	2 = "CN",	// チャージノート。終端ノートの判定タイミングで対応するボタンを離す必要がある。2方向入力が可能なデバイスの場合は「逆向き」の入力をもって「離した」と判定してもよい
	3 = "HCN",	// チャージノートの仕様に加えて、押している間は一定間隔で判定が発生し続ける。具体的な間隔と回復/ダメージ量は解釈器が決定する
}

// HeaderInfo.difficulty の推奨解釈
enum Difficulty
{
	0 = "Unknown",	// 不明もしくは難易度未設定
	1 = "Easy" = "Beginner",
	2 = "Normal" = "Standard",
	3 = "Hard" = "Advanced" = "Hyper",
	4 = "Extreme" = "Another" = "EX",
	5 = "Insane" = "Master" = "Leggendaria",
}

// HeaderInfo.level の推奨解釈
// 0 は「1より簡単」という意味ではなく、特殊なレベル(未指定、測定不能など)を表す。解釈器は小数点以下の値を無視してもよい
// beat系	: 1 - 12
// popn系	: 1 - 50

object Audio
{
	"filename"	: string,	// 参照するファイル名。異なるディレクトリを参照する場合は、"/"で区切った相対パスにする。

	"offset"	: Rational,	// 演奏開始位置(秒)
	"length"	: Rational,	// 演奏時間(秒)。0以下の場合は末尾からの時間を表す

	"volume": number,	// 再生音量。元々の音声の音量を 1 とした小数値で表す。省略時は 1
	"pan"	: number,	// 定位。左端を-1、右端を1とした小数値で表す。解釈器はこれを無視してもよい
	"pitch"	: number,	// cent単位で表されるピッチ。offset及びlengthの値はこの値の影響を受けず、元々の音声における時間位置を表す。解釈器はこれを無視してもよい
}

object Image
{
	"filename"	: string,	// 参照するファイル名。異なるディレクトリを参照する場合は、"/"で区切った相対パスにする。

	"offset"	: Rational,	// 対象が動画である場合、再生の開始位置(秒)。静止画の場合は無視する
	"length"	: Rational,	// 対象が動画である場合、再生する長さ(秒)。0以下の場合は末尾からの時間を表す。静止画の場合は無視する

	"cx"	: number,	// 切り抜き範囲の左端(pixel)。0未満の場合、不足分は透明ピクセルで補う
	"cy"	: number,	// 切り抜き範囲の上端(pixel)。0未満の場合、不足分は透明ピクセルで補う
	"cw"	: number,	// 切り抜き範囲の幅(pixel)。0以下の場合は右端からの幅を表す。cxと合わせて元々の幅を超える場合、不足分は透明ピクセルで補う
	"ch"	: number,	// 切り抜き範囲の高さ(pixel)。0以下の場合は下端からの高さを表す。cyと合わせて元々の高さを超える場合、不足分は透明ピクセルで補う
}

object Conductor
{
	"y"		: Position,	// 時間位置。同じ時刻に複数の指揮者オブジェクトが存在する場合、記述された順に適用する(結果として最後に記述されたもののみが適用される)
	"bpm"	: number,	// テンポ(bpm)。省略時は0ではなく、この時刻ではテンポを変更しないことを表す
	"stop"	: Rational,	// 譜面の停止期間(time_base依存)。同じ時刻にbpmとstopが存在する場合、先にbpmを処理する。負の値を指定した場合はワープになる(間にあるオブジェクトは全て無視する)
	"scroll"	: number,	// スクロール速度。解釈器の通常のスクロール速度を 1 とした割合で表す。省略時は0ではなく、この時刻ではスクロール速度を変更しないことを表す
}

object Note
{
	"y"	: Position, // 時間位置。同じ位置かつ同じレーンに複数のノートが存在する場合、解釈器は一度の入力でその全てを処理しなければならない
	"x"	: number,	// 演奏レーン。1以上の場合、具体的にどのレーンがどの位置に対応するかは解釈器が header.chart_type を元に決定する。0以下の場合はBGM扱い(総ノート数に計上しない)

	"i"	: number,	// sounds のキーに対応したリソースID。無効なID(省略時を含む)の場合、いわゆる無音ノートになる
	"t"	: Enum,		// ノートタイプ(後述)
	"g" : number,	// 最高判定で処理した場合のゲージ回復量。省略時は header.total に従う。負の場合は地雷になり、解釈器はノートの見た目を変えることが推奨される

	"lt": Enum,		// ロングノート形式。終端(t=1)となるノートで指定する。省略時は header.ln_type に従う
	"s"	: Rational,	// 演奏開始位置(秒)。soundsでの指定とは合算する。解釈器は後述の特殊値を受け入れなければならない
	"volume": number,	// 再生音量。元々の音声の音量を 1 とした小数値で表す。soundsでの指定とは乗算する
	"pan"	: number,	// 定位。左端を-1、右端を1とした小数値で表す。soundsでの指定とは合算する。解釈器はこれを無視してもよい
	"pitch"	: number,	// cent単位で表されるピッチ。soundsでの指定とは合算する。解釈器はこれを無視してもよい
}

// Note.t の解釈
// 明記されたものはMUSTであり、解釈器はこれ以外の解釈をしてはならない
enum NoteType
{
	0 = "Normal",		// 通常のノート
	-1 = "Invisible",	// 不可視ノート。総ノート数/判定/評価に影響せず、自動再生を行わない。該当レーンのボタンを操作した場合にのみ音声を再生する
	1 = "LongEnd",		// ロングノート終端。該当レーンのボタンを離した場合に判定処理する。直前のノートが通常ノートではない場合はこのノートを無視する
	2 = "LongBody",		// ロングノート中間。該当レーンのボタンが押されている場合に判定処理する。直前のノートが通常またはロング中間ノートではない場合はこのノートを無視する
}

// Note.s の解釈
// 明記されたものはMUSTであり、解釈器はこれ以外の解釈をしてはならない
// 数値として解釈可能な場合は、その数値として扱う
// "c", "continue"	-> 直前に同じリソースIDのノートが存在する場合、演奏開始位置の指定は無視して、そのノートの「続き」から再生する。存在しない場合はsoundsでの指定に従い「最初」から再生する
// "s", "stop"		-> 直前に同じリソースIDのノートが存在する場合、そのノートによって再生された音声を停止する。存在しない場合は無音ノートとして扱う

object MetaNote
{
	"y"	: Position, // 時間位置。同じ位置かつ同じレーンに複数のノートが存在する場合、記述された順に適用する(結果として最後に記述されたもののみが適用される)
	"x"	: number,	// チャンネル。推奨値は後述するが、省略時の値も含め、解釈器はこれに従わなくてもよい

	"i"	: number,	// images のキーに対応したリソースID。無効なID(省略時を含む)の場合、該当レーンの映像を消去する
	"v"	: string,	// そのノート固有の値。その処理方法は解釈機に完全に委ねられる

	// 以下のプロパティは映像ノートでのみ有効
	"cx"	: number,	// 切り抜き範囲の左端(pixel)。imagesでの指定とは合算する
	"cy"	: number,	// 切り抜き範囲の上端(pixel)。imagesでの指定とは合算する
	"cw"	: number,	// 切り抜き範囲の幅(pixel)。imagesでの指定とは合算する
	"ch"	: number,	// 切り抜き範囲の高さ(pixel)。imagesでの指定とは合算する

	"dx"	: number,	// 描画位置(pixel)。描画領域の中心を 0 とする
	"dy"	: number,	// 描画位置(pixel)。描画領域の中心を 0 とする
	"ox"	: number,	// 描画基準位置(pixel)。省略時は切り抜き後の幅の 1/2 (小数点以下切り捨て)
	"oy"	: number,	// 描画基準位置(pixel)。省略時は切り抜き後の高さの 1/2 (小数点以下切り捨て)
	"angle"	: number,	// 画像の回転角度(°)。時計回りで指定
	"ax"	: number,	// 回転中心(pixel)。省略時は ox に等しいと見なす
	"ay"	: number,	// 回転中心(pixel)。省略時は oy に等しいと見なす
}

// MetaNote.x の推奨解釈
// 1 -> BGA(ベース)
// 2 -> BGA(レイヤー)
// 3 -> BGA(ミスレイヤー)

object FlowInfo
{
	"index"	: string,	// 分岐インデックスを決定する構文(後述)。省略された場合はこの分岐全体を無視する
	"branches"	: FlowBranch{},	// 分岐インデックス(0 以上の整数)をキーとする分岐枝の連想配列
	"default"	: FlowBranch,	// branches に対応する分岐インデックスが存在しなかった場合に選択される分岐枝
}

object FlowBranch
{
	"data_id"	: number,		// 解釈器が読み込むべき Bm3onData.flow_data のID
	"flows"		: FlowInfo[],	// ネストした分岐情報のリスト。分岐そのものは無制限(※)にネスト可能だが、header, flows, flow_data は常にルートデータのものを参照する
}

// FlowInfo.index の記法
// 明記されたものはMUSTであり、解釈器はこれ以外の解釈をしてはならない
// これ以外の形式(例えば現在のスコアに応じた動的な分岐など)については、解釈器が独自に実装してもよい
// number	-> 常にその値をその分岐インデックスとする
// "rand(X)"	-> 0 以上 X 未満のランダムな整数。乱数の供給方法は解釈器が任意に実装してよい
// "random(X,Y)"	-> X 以上 Y 未満のランダムな整数
